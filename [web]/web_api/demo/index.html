Use it in a resource

<!-- meta.xml -->
<include resource="web_api" />
<html src="web/http.html" default="true" />
<file src="web/index.html" type="client" />
<export function="myCall" type="server" http="true" />
<!-- web/http.html: read the page and the files
     it links, web_api inlines them -->
<* local PAGE = "web/index.html"
   local function read(p) ... end
   local html = read(PAGE)
   local files = {}
   for _, p in ipairs(exports.web_api:getAssetRefs(html, PAGE)) do
     files[p] = read(p)
   end
   httpWrite(exports.web_api:renderPage(html, files, PAGE)) *>
<!-- web/index.html -->
<link rel="stylesheet" href="/web_api/osa.css">
<script src="/web_api/osa.js"></script>
<script>
  const api = OSA.api();
  const r = await api.call('myCall', 1, 2);
</script>
-- server, only for an in-game site
exports.web_api:registerApp({
  title = "My app", site = "my-app.eu",
  category = "services", description = "..." })
